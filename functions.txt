functions.txt — exported surface of the chat + MCP layers (path :: symbol — purpose)

src/chat/types.ts
  :: ChatRole, TraceStatus                 union types ("user"|"assistant", "running"|"done"|"error")
  :: ToolTrace                             one MCP invocation rendered in an assistant turn (id, toolId, serverId, name, arguments, status, signature?, output?, error?, startedAt, endedAt?)
  :: ChatMessageRecord                     persisted message (content, thought?, traces?)
  :: ChatSession                           session with messages[]
  :: ChatSessionSummary                    sidebar row (id, title, model, messageCount, updatedAt)
  :: ChatRunRequest                        run payload (sessionId?, message, model, systemPrompt?, tools?, temperature?, reasoningEffort?, maxToolRounds?)
  :: ChatStreamEvent                       SSE contract: session|round|token|thought|tool_start|tool_end|message|usage|error|done

src/chat/store.ts
  :: DEFAULT_TITLE, MAX_TITLE_LENGTH       "New conversation"; 120-char rename cap
  :: deriveTitle(content)                   first-user-message title (64 chars max)
  :: ChatSessionStore(dir = getChatDataDir())  in-memory cache + atomic per-session write to data/chats/<id>.json
  :: #create(model, systemPrompt?) / #get(id) / #list() / #appendMessage(id, record) / #update(id, mutate)
  :: #rename(id, title)                    whitespace-collapsed rename; undefined on blank/overside/unknown id
  :: #rewindTo(id, index)                   drops records from `index` onwards (edit / regenerate); undefined when out of range
  :: #remove(id)                           deletes the session and its file

src/chat/history.ts
  :: MAX_HISTORY_RECORDS                    transcript window fed upstream
  :: toOpenAiMessages(records)              persisted records -> OpenAI messages (assistant traces replay as tool_calls + role:"tool" results, signatures restored)
  :: trimHistory(messages, limit?)          head-trim preserving the leading user turn
  :: stringifyArguments(value)              safe JSON.stringify for replayed tool arguments

src/chat/agent-loop.ts
  :: DEFAULT_MAX_TOOL_ROUNDS               6 (clamped by maxToolRounds)
  :: runAgentTurn(deps, params, emit, signal?)  multi-round loop: map -> stream -> execute tools -> REPLAY assistant tool_calls + results -> final text; persists the assistant record
  :: executeToolCall(registry, call, emit, signal?) registry lookup ("server__tool"), runs tool, emits tool_start/tool_end once

src/chat/events.ts
  :: SseChannel                             SSE framing for ChatStreamEvent (send/close); swallows writes after client disconnect

src/mcp/types.ts
  :: McpServerConfig / transport kinds      stdio | http | streamable-http | inprocess + include/exclude filters + timeoutMs
  :: McpToolInfo, ServerStatus shapes

src/mcp/config.ts
  :: CONFIG_FILENAME, DEFAULT_CALL_TIMEOUT_MS
  :: expandEnv(value)                        ${VAR} expansion from process.env
  :: parseMcpConfig(raw, log)               validate + normalise (drops invalid servers, never throws on missing file)
  :: resolveConfigPath(explicit?)           --mcp-config -> PROXY_MCP_CONFIG -> ./mcp.config.json -> ~/.gemini/mcp.config.json
  :: loadMcpConfig(path?, log)              read file then parse; null config = warning only

src/mcp/connect.ts
  :: CLIENT_INFO, CONNECT_TIMEOUT_MS       client metadata for the MCP handshake

src/mcp/expression.ts

  :: evaluateExpression(source)             shunting-yard evaluator: + - * / % ^, parentheses, unary +/-, constants pi/e/tau, functions abs sqrt cbrt floor ceil round sin cos tan asin acos atan exp log log10 log2 pow min max atan2; no eval

src/mcp/builtin-server.ts

  :: McpToolRegistry.empty(log)             builtin-only registry
